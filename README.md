## Sobre mim

```java
import java.util.List;

public final class Gisely implements Developer {

    private final String apelido;
    private final String curso;
    private final String foco;
    private final List<String> stackEmEstudo;
    private final boolean sempreAprendendo;

    public Gisely() {
        this.apelido = "Gisa";
        this.curso = "Engenharia de Software";
        this.foco = "Desenvolvimento Backend";
        this.stackEmEstudo = List.of("Full Stack", "Java", "SQL");
        this.sempreAprendendo = true;
    }

    @Override
    public void apresentar() {
        System.out.printf("Estudante de %s%n", curso);
        System.out.printf("Foco atual: %s%n", foco);
        System.out.printf("Stack em estudo: %s%n", String.join(", ", stackEmEstudo));
        System.out.printf("Sempre aprendendo: %s%n", sempreAprendendo);
    }

    public static void main(String[] args) {
        new Gisely().apresentar();
    }
}

interface Developer {
    void apresentar();
}
```
